cmake_minimum_required(VERSION 3.10)

project(lecteurAudio C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_executable(LecteurAudio
    src/main.c
    src/renderer.c
    src/call_OS.c
    src/audio.c
)

target_include_directories(LecteurAudio PRIVATE
    libs/SDL2-2.32.8/i686-w64-mingw32/include
    libs/SDL2_image-devel-2.8.8-mingw/SDL2_image-2.8.8/i686-w64-mingw32/include
    libs/SDL2_mixer/include
    libs/libsndfile/include
)

target_link_directories(LecteurAudio PRIVATE
    libs/SDL2-2.32.8/i686-w64-mingw32/lib
    libs/SDL2_image-devel-2.8.8-mingw/SDL2_image-2.8.8/i686-w64-mingw32/lib
    libs/SDL2_mixer/lib
    libs/libsndfile/lib
)

target_link_libraries(LecteurAudio PRIVATE
    mingw32
    SDL2main
    SDL2
    SDL2_image
    SDL2_mixer
    sndfile
)
